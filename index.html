    Repository Automation & Deployment Guide

    Commit the three files below to github.com/fastmail7009-netizen/pdf-library so every .pdf pushed anywhere in the repository automatically appears in this read-only SDS binder.

    Zero Manual Maintenance

    1Enable Actions Write Access

    In repository Settings → Actions → General → Workflow permissions, select Read and write permissions so the workflow can commit pdf-manifest.json on push to main.

    2Commit Workflow, JS & Starter JSON

    Add .github/workflows/sync-pdf-library.yml, pdf-library.js, and pdf-manifest.json alongside your index.html and optional logo.png.

    3Push Any .pdf Anywhere in Repo

    Upload PDFs to root or subfolders (e.g. solvents/Acetone.pdf). GitHub Actions scans all .pdf files, rebuilds pdf-manifest.json, and GitHub Pages serves the updated binder immediately. 

    name: Sync PDF Library Manifest

on:
  push:
    branches:
      - main
  workflow_dispatch:

permissions:
  contents: write

jobs:
  build-pdf-manifest:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout repository
        uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - name: Scan all PDF files and generate pdf-manifest.json
        run: |
          python3 - << 'EOF'
          import datetime
          import json
          import os
          import re
          import subprocess
          import urllib.parse

          repo = os.environ.get("GITHUB_REPOSITORY", "fastmail7009-netizen/pdf-library")
          branch = os.environ.get("GITHUB_REF_NAME", "main")
          sheets = []

          for root, dirs, files in os.walk("."):
              dirs[:] = sorted(d for d in dirs if not d.startswith("."))
              for fname in sorted(files):
                  if not fname.lower().endswith(".pdf"):
                      continue
                  rel_path = os.path.relpath(os.path.join(root, fname), ".").replace("\\", "/")
                  folder = os.path.dirname(rel_path) or "Root"
                  stem = os.path.splitext(fname)[0]

                  # Extract leading SDS code if present, otherwise generate deterministic code
                  code_match = re.match(r"^([A-Z]{2,6}[-_][A-Z0-9_-]{2,32})[\s_-]+(.+)$", stem, re.I)
                  if code_match:
                      code = code_match.group(1).upper()
                      raw_title = code_match.group(2)
                  else:
                      slug = re.sub(r"[^A-Za-z0-9]+", "-", stem).strip("-").upper()
                      code = f"SDS-{slug[:24]}" if slug else "SDS-DOC"
                      raw_title = stem

                  title = re.sub(r"[-_]+", " ", raw_title).strip()
                  size_bytes = os.path.getsize(rel_path)

                  if size_bytes >= 1024 * 1024:
                      size_str = f"{size_bytes / (1024 * 1024):.2f} MB"
                  else:
                      size_str = f"{max(1, round(size_bytes / 1024))} KB"

                  git_date = subprocess.check_output(
                      ["git", "log", "-1", "--format=%cs", "--", rel_path],
                      text=True,
                  ).strip()
                  updated_at = git_date or datetime.date.today().isoformat()

                  encoded_url = "./" + "/".join(urllib.parse.quote(p) for p in rel_path.split("/"))

                  sheets.append({
                      "code": code,
                      "title": title,
                      "filename": fname,
                      "path": rel_path,
                      "folder": folder,
                      "sizeBytes": size_bytes,
                      "size": size_str,
                      "updatedAt": updated_at,
                      "url": encoded_url,
                  })

          sheets.sort(key=lambda s: (s["folder"].lower(), s["code"].lower(), s["filename"].lower()))

          manifest = {
              "repository": repo,
              "branch": branch,
              "generatedAt": datetime.datetime.now(datetime.timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ"),
              "totalCount": len(sheets),
              "sheets": sheets,
          }

          with open("pdf-manifest.json", "w", encoding="utf-8") as f:
              json.dump(manifest, f, indent=2)
              f.write("\n")

          print(f"Generated pdf-manifest.json with {len(sheets)} PDF(s).")
          EOF

      - name: Commit and push pdf-manifest.json if changed
        run: |
          git config user.name "github-actions[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          git add pdf-manifest.json
          if git diff --cached --quiet; then
            echo "No changes to pdf-manifest.json."
          else
            git commit -m "chore: sync pdf-manifest.json [skip ci]"
            git push
          fi

 * pdf-library.js — Read-Only SDS Binder Loader for fastmail7009-netizen/pdf-library
 * Fetches ./pdf-manifest.json generated by GitHub Actions on push to main.
 */
(function (global) {
  'use strict';

  var MANIFEST_URL = './pdf-manifest.json';

  function normalizeManifestPayload(payload) {
    var rawList = Array.isArray(payload)
      ? payload
      : (payload && (payload.sheets || payload.files || payload.documents)) || [];

    return rawList.map(function (entry, idx) {
      var relPath = String(entry.path || entry.filename || ('document-' + (idx + 1) + '.pdf'));
      var filename = entry.filename || relPath.split('/').pop();
      var folder = entry.folder || (relPath.indexOf('/') !== -1 ? relPath.slice(0, relPath.lastIndexOf('/')) : 'Root');
      var code = entry.code || ('SDS-' + String(idx + 1).padStart(3, '0'));
      var title = entry.title || filename.replace(/\.pdf$/i, '').replace(/[-_]+/g, ' ');
      var url = entry.url || ('./' + relPath.split('/').map(encodeURIComponent).join('/'));

      return {
        code: code,
        title: title,
        filename: filename,
        path: relPath,
        folder: folder,
        size: entry.size || 'PDF',
        sizeBytes: Number(entry.sizeBytes || 0),
        updatedAt: entry.updatedAt || 'Unknown',
        url: url,
        signalWord: entry.signalWord || 'HAZARD INFO IN PDF',
        cas: entry.cas || 'See Full PDF',
        ghs: Array.isArray(entry.ghs) ? entry.ghs : ['Refer to Full SDS Document'],
        ppe: entry.ppe || 'Inspect Section 8 of attached PDF before handling.',
        firstAid: entry.firstAid || 'Inspect Section 4 of attached PDF for emergency measures.',
        storage: entry.storage || 'Inspect Section 7 of attached PDF for storage requirements.',
        nfpa: entry.nfpa || null
      };
    });
  }

  async function fetchSdsManifest(fallbackSheets) {
    try {
      var response = await fetch(MANIFEST_URL + '?v=' + Date.now(), { cache: 'no-store' });
      if (!response.ok) {
        throw new Error('HTTP ' + response.status);
      }
      var data = await response.json();
      var sheets = normalizeManifestPayload(data);
      return {
        source: 'live',
        generatedAt: data.generatedAt || null,
        repository: data.repository || 'fastmail7009-netizen/pdf-library',
        sheets: sheets
      };
    } catch (err) {
      return {
        source: 'fallback',
        error: err && err.message ? err.message : 'Manifest unavailable',
        generatedAt: null,
        repository: 'fastmail7009-netizen/pdf-library',
        sheets: normalizeManifestPayload(fallbackSheets || [])
      };
    }
  }

  global.PdfLibraryLoader = {
    fetchSdsManifest: fetchSdsManifest,
    normalizeManifestPayload: normalizeManifestPayload
  };
})(window);
